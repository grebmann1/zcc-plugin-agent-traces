import {cpSync,mkdirSync,readFileSync,writeFileSync} from 'node:fs';
import {join,basename,resolve} from 'node:path';
import {execFile} from 'node:child_process';
import {promisify} from 'node:util';
import {test,expect} from './fixtures/app.js';
test.use({launchEnv:{ZCC_FAKE_PROVIDER:'1',ZCC_SKIP_PLUGIN_NPM:'1'},initialConfig:{sponsorPromptDismissed:true},isolateBundledCatalog:true});
test('optional ports install and work through the built Electron UI',async({app,home},testInfo)=>{
 test.setTimeout(180000);
 const page=app.window;
 const cli=resolve(process.env.ZCC_E2E_APP_ROOT??'.','packages/cli/dist/bin/zcc');
 const env={...process.env,HOME:home,ZCC_DATA_DIR:join(home,'.zcc'),ZCC_SERVER_URL:new URL(page.url()).origin,ZCC_SESSION_ID:undefined,ZCC_SESSION_TOKEN:undefined,ZCC_SKIP_PLUGIN_NPM:'1',npm_config_registry:'https://registry.npmjs.org/'};
 const run=(args:string[],cwd:string)=>promisify(execFile)(process.execPath,[cli,...args],{cwd,env,timeout:120000,maxBuffer:2*1024*1024});
 const copies:Record<string,string>={};
 for(const id of ['agent-traces','office-viewer','burn']){
  const target=join(home,'plugins-source',id);mkdirSync(target,{recursive:true});
  cpSync(join(process.env.BB_PORTS_ROOT!,id),target,{recursive:true,filter:p=>!['node_modules','.git','coverage'].includes(basename(p))});copies[id]=target;
  await run(['plugin','install',target],target);
  await expect.poll(()=>page.evaluate(async id=>(await window.cc.pluginApps.list()).find(p=>p.id===id)?.status,id)).toBe('running');
 }
 const projectPath=join(home,'ports-project');mkdirSync(projectPath);
 // Genuine zipped Office fixtures generated by the plugin's pure parser tests.
 for(const name of ['report.docx','forecast.xlsx','briefing.pptx'])writeFileSync(join(projectPath,name),Buffer.from(readFileSync(join(process.env.BB_PORTS_ROOT!,'office-viewer','integration',name+'.base64'),'utf8'),'base64'));
 const ids=await page.evaluate(async root=>{
  const project=await window.cc.projects.add(root);if(!project.ok)throw new Error(project.message);
  const response=await fetch('/api/v1/threads',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({projectId:project.value.id,providerId:'fake',input:'Port verification call_tool:execution_checks_list report_tool_result'})});
  const body=await response.json();if(!response.ok)throw new Error(JSON.stringify(body));return {threadId:(body.thread??body.value).id as string,projectId:project.value.id};
 },projectPath);
 await expect.poll(()=>page.evaluate(async id=>(await (await fetch(`/api/v1/threads/${id}`)).json()).thread.status,ids.threadId)).toBe('idle');
 async function navigate(path:string){await page.evaluate(path=>{history.pushState({},'',path);dispatchEvent(new PopStateEvent('popstate'))},path)}
 await navigate('/plugins/agent-traces/main');
 await expect(page.getByRole('heading',{name:'Agent Traces',exact:true})).toBeVisible();
 await page.getByLabel('Thread',{exact:true}).selectOption(ids.threadId);
 await expect(page.getByText(/recorded events ·/)).toBeVisible();
 await expect(page.getByLabel('Filter steps')).toBeVisible();
 const trace=await page.evaluate(async id=>window.cc.pluginApps.callRpc('agent-traces','trace',{threadId:id}),ids.threadId);
 expect((trace as {eventCount:number}).eventCount).toBeGreaterThan(0);
 await page.getByLabel('Filter steps').fill('no matching step');await expect(page.getByText('No matching recorded steps.')).toBeVisible();
 await page.screenshot({path:testInfo.outputPath('agent-traces.png')});
 await navigate('/plugins/burn/main');await expect(page.getByRole('heading',{name:'Burn',exact:true})).toBeVisible();
 await page.getByLabel('input',{exact:true}).fill('3');await page.getByLabel('cached',{exact:true}).fill('0.3');await page.getByLabel('output',{exact:true}).fill('15');await page.getByLabel('budget',{exact:true}).fill('2');
 await page.getByRole('button',{name:'Save rates',exact:true}).click();await expect(page.getByText('Rates saved')).toBeVisible();
 await page.getByLabel('Thread',{exact:true}).selectOption(ids.threadId);await expect(page.getByText(/Estimate unavailable|\$[0-9]+\.[0-9]{4}/)).toBeVisible();
 expect(await page.evaluate(()=>window.cc.pluginApps.callRpc('burn','rates',{}))).toEqual({input:3,cached:.3,output:15,budget:2});
 await page.screenshot({path:testInfo.outputPath('burn.png')});
 await navigate(`/threads/${ids.threadId}`);await expect(page.getByTestId('thread-detail')).toBeVisible();
 for(const [file,expected] of [['report.docx','Business proposal'],['forecast.xlsx','Quarter'],['briefing.pptx','Delivery overview']]){
  expect(await page.evaluate(async({id,file})=>(await fetch(`/api/v1/threads/${id}/open`,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({file:{source:'workspace',path:file}})})).status,{id:ids.threadId,file})).toBe(200);
  const panel=page.getByTestId('thread-secondary-panel');await expect(panel.getByText(expected,{exact:true})).toBeVisible();
  await panel.getByRole('button',{name:'Quote section',exact:true}).click();await expect(panel.getByText('Quote added to the conversation draft')).toBeVisible();
  await expect(page.getByTestId('thread-command-input').first()).toContainText(expected);
 }
 await page.screenshot({path:testInfo.outputPath('office-viewer.png')});
 const denied=await page.evaluate(async ids=>{try{await window.cc.pluginApps.callRpc('office-viewer','preview',{path:'/etc/passwd.docx',source:{kind:'workspace',projectId:ids.projectId,threadId:ids.threadId,environmentId:null}});return false}catch{return true}},ids);expect(denied).toBe(true);
 // A visible source edit must update the already-open plugin after reload.
 await navigate('/plugins/agent-traces/main');const appSource=join(copies['agent-traces'],'app.tsx');writeFileSync(appSource,readFileSync(appSource,'utf8').replace('title="Agent Traces"','title="Agent Traces reload verified"'));
 
 await run(['plugin','reload','agent-traces'],copies['agent-traces']);
 await expect(page.getByRole('heading',{name:'Agent Traces reload verified'})).toBeVisible();
 expect(await page.evaluate(()=>window.cc.pluginApps.callRpc('burn','rates',{}))).toEqual({input:3,cached:.3,output:15,budget:2});
});
